! RT-EDGE (Cisco 2911) - comandos aplicados
! Para um registro fiel, substitua pelo resultado de "show running-config"
enable
configure terminal
hostname RT-EDGE
no ip domain-lookup

! ---- Subinterfaces (gateway de cada VLAN) ----
interface gi0/0
 no shutdown
interface gi0/0.10
 description GW-TI
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip nat inside
interface gi0/0.20
 description GW-RH
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip nat inside
interface gi0/0.30
 description GW-FINANCEIRO
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 ip nat inside
interface gi0/0.40
 description GW-SERVIDORES
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.0
 ip nat inside

! ---- Link com a "internet" ----
interface gi0/1
 description LINK-ISP
 ip address 198.51.100.2 255.255.255.252
 ip nat outside
 no shutdown

! ---- Rota padrao ----
ip route 0.0.0.0 0.0.0.0 198.51.100.1

! ---- DHCP ----
ip dhcp excluded-address 192.168.10.1 192.168.10.10
ip dhcp excluded-address 192.168.20.1 192.168.20.10
ip dhcp excluded-address 192.168.30.1 192.168.30.10

ip dhcp pool TI
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 192.168.40.10
 domain-name ricarte.local
ip dhcp pool RH
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 192.168.40.10
 domain-name ricarte.local
ip dhcp pool FINANCEIRO
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 192.168.40.10
 domain-name ricarte.local

! ---- NAT (PAT) ----
ip access-list standard NAT-INTERNO
 permit 192.168.10.0 0.0.0.255
 permit 192.168.20.0 0.0.0.255
 permit 192.168.30.0 0.0.0.255
 permit 192.168.40.0 0.0.0.255
exit
ip nat inside source list NAT-INTERNO interface gi0/1 overload

! ---- ACL: RH nao acessa o Financeiro ----
ip access-list extended BLOQUEIA-RH-FIN
 deny ip 192.168.20.0 0.0.0.255 192.168.30.0 0.0.0.255
 permit ip any any
exit
interface gi0/0.20
 ip access-group BLOQUEIA-RH-FIN in
end
copy running-config startup-config
